from fastapi import FastAPI from fastapi.middleware.cors import CORSMiddleware from pydantic import BaseModel import sys import os sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__)))) from core_engine import State, Session, ScienceClient, _download_games, _parse_games, select_games app = FastAPI() app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"], ) class RunRequest(BaseModel): token: str cookie: str hours: float = 10.0 count: int = 50 @app.post("/api/run") def run_bot(req: RunRequest): try: state = State() state.token = req.token state.cookie = req.cookie session = Session.new() client = ScienceClient(state, session) try: raw = _download_games() games = _parse_games(raw) except Exception: return {"success": False, "message": "Không thể tải danh sách game"} selected = select_games(games, state, req.count) events = [] for game in selected: events.extend(client.build_session(game, int(req.hours * 3600 * 1000))) status = client.post(events) if status == 204: return {"success": True, "message": f"Đã gửi thành công {len(selected)} games."} else: return {"success": False, "message": f"Lỗi Discord API: HTTP {status}. Token/Cookie có thể đã hết hạn."} except Exception as e: return {"success": False, "message": str(e)}
